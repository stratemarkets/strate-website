      Stacked footprint imbalances

      Marks diagonal imbalances in each bar's footprint, and draws a zone where several stack up. A zone that later holds is a common order-flow reference.

      StrateScript
      indicator "Stacked imbalances" { overlay: true }

ratio = input.float("Ratio", 3, min: 1.5, step: 0.5)
minVol = input.int("Min volume on the strong side", 10, min: 0)
stack = input.int("Min stacked rows", 3, min: 2)

fp = bar.footprint
for p in fp.imbalances("buy", ratio: ratio, min_volume: minVol) {
  cell(p, "", side: right, bg: #26a69a60)
}
for p in fp.imbalances("sell", ratio: ratio, min_volume: minVol) {
  cell(p, "", side: left, bg: #ef535060)
}
for z in fp.stacked("buy", ratio: ratio, count: stack) {
  box.new(bar.time, z[1], bar.time + 30m, z[0], color: #26a69a20, border: #26a69a, fade: "right")
}
for z in fp.stacked("sell", ratio: ratio, count: stack) {
  box.new(bar.time, z[1], bar.time + 30m, z[0], color: #ef535020, border: #ef5350, fade: "right")
}

      What to notice:

      • a buy imbalance at a price compares the buy volume there with the sell volume one row lower (the diagonal); a sell imbalance compares the sell volume with the buy volume one row higher;
      • fp.stacked returns [low, high] pairs; the boxes extend 30 minutes to the right and fade out.
